Doodle Glitch Effect Demo

Click buttons above to change intensity

.doodle-glitch {
  animation: doodle-jitter 0.5s infinite steps(1);
}

@keyframes doodle-jitter {
  0% { transform: translate(0, 0) rotate(0deg); }
  20% { transform: translate(-1px, 1px) rotate(-0.5deg); }
  40% { transform: translate(1px, -1px) rotate(0.5deg); }
  60% { transform: translate(-1px, -1px) rotate(-0.3deg); }
  80% { transform: translate(1px, 0.5px) rotate(0.2deg); }
}

/* The steps(1) makes it jump between frames */
